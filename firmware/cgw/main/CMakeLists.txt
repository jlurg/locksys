# main: composition root (app_main, static tasks and queues, port bindings).
# ls_version_gen.h is generated into the build directory from VERSION and git (not committed).

idf_component_register(
    SRCS cgw_main.c cgw_wiring.c
    INCLUDE_DIRS . ${CMAKE_CURRENT_BINARY_DIR}
    PRIV_REQUIRES cgw_core cgw_com cgw_ports cgw_platform_esp cgw_wifi_esp cgw_ws_esp
                  cgw_can_esp cgw_crypto_psa cgw_store_nvs cgw_hmi_esp cgw_pairing ls_common ls_e2e
                  bootloader_support esp_event esp_system esp_app_format log
)

include(${CMAKE_CURRENT_LIST_DIR}/../cmake/cgw_component.cmake)
cgw_component_warnings(adapter)

# Build identification (ADR 0006): SemVer from VERSION, short hash and dirty flag from git.
file(STRINGS ${CMAKE_CURRENT_LIST_DIR}/../../../VERSION ls_version LIMIT_COUNT 1)
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" _ "${ls_version}")
set(LS_VER_MAJOR ${CMAKE_MATCH_1})
set(LS_VER_MINOR ${CMAKE_MATCH_2})
set(LS_VER_PATCH ${CMAKE_MATCH_3})
set(LS_GIT_HASH "0")
set(LS_GIT_DIRTY 0)
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --short=7 HEAD
        WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR} OUTPUT_VARIABLE git_hash
        RESULT_VARIABLE git_rc OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(git_rc EQUAL 0 AND git_hash MATCHES "^[0-9a-f]+$")
        set(LS_GIT_HASH "${git_hash}")
        execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain --untracked-files=no
            WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR} OUTPUT_VARIABLE git_status
            RESULT_VARIABLE git_rc ERROR_QUIET)
        if(git_rc EQUAL 0 AND NOT git_status STREQUAL "")
            set(LS_GIT_DIRTY 1)
        endif()
    endif()
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ls_version_gen.h.tmp
"/* DO NOT EDIT — generated by firmware/cgw/main/CMakeLists.txt from VERSION and git */
#ifndef LS_VERSION_GEN_H
#define LS_VERSION_GEN_H
#define LS_VERSION_STRING \"${ls_version}\"
#define LS_VERSION_MAJOR (${LS_VER_MAJOR}u)
#define LS_VERSION_MINOR (${LS_VER_MINOR}u)
#define LS_VERSION_PATCH (${LS_VER_PATCH}u)
#define LS_GIT_HASH (0x${LS_GIT_HASH}u)
#define LS_GIT_DIRTY (${LS_GIT_DIRTY}u)
#endif
")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/ls_version_gen.h.tmp
    ${CMAKE_CURRENT_BINARY_DIR}/ls_version_gen.h COPYONLY)
